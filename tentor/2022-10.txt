data DTree a
= Decision a
| Question String (DTree a) (DTree a)
deriving (Eq, Show)


mapDecision :: (a -> b) -> DTree a -> DTree b
mapDecision f (Decision a) = (Decision (f a))
mapDecision f (Question str l r) = (Question str (mapDecision f l) (mapDecision f r))






takeDecision :: DTree a -> IO a
takeDecision t = case t of
    (Decision a)          -> do return a
    (Question str yes no) -> do
        putStr (str ++ "yes / no")
        answer <- getLine
        if answer == "yes" then takeDecision yes else takeDecision no













type Pixel = (Int, Int)

data BillBoard = BB { size :: (Int, Int), actives :: [Pixel] }

inactives :: Billboard -> Int
inactives (BB (x,y) ac) = x * y - (length ac)


invert :: Billboard -> Billboard
invert (BB (x,y) ac) = [(x,y) | x <- [0..x], y <- [0..y], not (elem (x,y) ac)  ]

type HTML = [Tag]


data Tag
    = Text String
    | Open String
    | Close String
    deriving (Eq, Show)


render :: HTML -> String
render html = go html
    where
        go [] = ""
        go (x:xs) = case x of
            (Text a)  -> a ++ go xs
            (Open a)  -> '<' : a ++ ">" ++ go xs
            (Close a) -> '<' : '/' : a ++ ">" ++ go xs




checkHtml :: HTML -> Bool
checkHtml html = go [] html
    where
        go xs []     = True
        go xs (y:ys) = case y of 
            (Text a)  -> go xs ys
            (Open a)  -> go (a:xs) ys
            (Close a) -> if (last xs) == a then go (init xs) ys else False
